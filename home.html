quickstart.js 
const response = await fetch(
  "https://api.nexora.ai/v1/chat", {
    method: "POST",
    headers: {
      "Authorization": `Bearer ${process.env.NEXORA_KEY}`,
      "Content-Type": "application/json"
    },
    body: JSON.stringify({
      model: "gpt-4o",
      messages: [{
        role: "user",
        content: "Hello, AI!"
      }]
    })
  }
);

const data = await response.json();
// → { content: "Hi! How can I help you today?" }

  apiKey: process.env.NEXORA_KEY

});
